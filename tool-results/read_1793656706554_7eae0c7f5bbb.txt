    33→// ============================================================
    34→// Session — opaque-token server sessions for admin auth.
    35→// ============================================================
    36→// Replaces the legacy "Bearer password" scheme. Login route
    37→// mints a random 32-byte token, stores it here, and sets it as
    38→// an httpOnly cookie. verifyAuth() reads the cookie → looks up
    39→// the row → returns the real User (id, email, role).
    40→//
    41→// userId is nullable to support the legacy password-gate mode
    42→// (anonymous admin session when no User row is linked). Real
    43→// logins always attach a userId.
    44→// ============================================================
    45→
    46→model Session {
    47→  id        String   @id @default(cuid())
    48→  token     String   @unique
    49→  userId    String?
    50→  user      User?    @relation(fields: [userId], references: [id], onDelete: Cascade)
    51→  /// Customer (storefront) user — set when method='customer'.
    52→  customerId String?
    53→  customer   Customer? @relation(fields: [customerId], references: [id], onDelete: Cascade)
    54→  /// Snapshot of email for display/audit when userId is null (password-gate).
    55→  email     String?
    56→  /// Snapshot of role at session-create time. One of:
    57→  /// super_admin | admin | staff | viewer | password_gate | customer
    58→  role      String   @default("viewer")
    59→  /// "nextauth" (admin user) | "password_gate" (legacy) | "customer" (storefront)
    60→  method    String   @default("nextauth")
    61→  expiresAt DateTime
    62→  createdAt DateTime @default(now())
    63→
    64→  @@index([userId])
    65→  @@index([customerId])
    66→  @@index([expiresAt])
    67→}

   486→// ============================================================
   487→// Coupon — discount code system (customer-input codes).
   488→// ============================================================
   489→// Dual-layer with ERPNext Pricing Rule:
   490→//   - Coupon (here): customer-input codes at checkout
   491→//     (WELCOME10, ONGKIR0, HEMAT20). Validated + redeemed here.
   492→//   - ERPNext Pricing Rule: automatic discounts per-item/customer
   493→//     (synced into Product price). Not managed here.
   494→//
   495→// type: "percent" | "fixed" | "free_shipping"
   496→//   percent       — value = 1..100 (percent off subtotal)
   497→//   fixed         — value = IDR amount off subtotal
   498→//   free_shipping — value ignored; sets shipping to 0
   499→//
   500→// scopeType: "all" | "category" | "product"
   501→//   all       — applies to any order
   502→//   category  — only when cart has items in scopeValue category
   503→//   product   — only when cart has item with scopeValue itemCode
   504→//
   505→// Anti-abuse:
   506→//   maxUsage     — total times coupon can be redeemed (0 = unlimited)
   507→//   maxPerUser   — per-customer limit (identified by email or phone)
   508→//   usedCount    — current redemption count (incremented atomically)
   509→// ============================================================
   510→
   511→model Coupon {
   512→  id          String   @id @default(cuid())
   513→  code        String   @unique
   514→  /// "percent" | "fixed" | "free_shipping"
   515→  type        String
   516→  /// Percent (1-100) or IDR amount. Ignored for free_shipping.
   517→  value       Int      @default(0)
   518→  /// Minimum subtotal (IDR) for coupon to apply. 0 = no minimum.
   519→  minOrder    Int      @default(0)
   520→  /// Max total redemptions. 0 = unlimited.
   521→  maxUsage    Int      @default(0)
   522→  /// Max redemptions per customer (by email or phone). 0 = unlimited.
   523→  maxPerUser  Int      @default(1)
   524→  usedCount   Int      @default(0)
   525→  startsAt    DateTime?
   526→  expiresAt   DateTime?
   527→  active      Boolean  @default(true)
   528→  /// "all" | "category" | "product"
   529→  scopeType   String   @default("all")
   530→  /// Category name or itemCode (when scopeType != "all")
   531→  scopeValue  String?
   532→  /// Optional internal note (admin only)
   533→  note        String?
   534→  createdAt   DateTime @default(now())
   535→  updatedAt   DateTime @updatedAt
   536→
   537→  usages      CouponUsage[]
   538→
   539→  @@index([active])
   540→  @@index([expiresAt])
   541→}

   606→// published: visibility on public storefront (hidden = admin-only)
   607→// order: manual ordering on storefront (asc)
   608→// ============================================================
   609→
   610→model Product {
   611→  id              String   @id @default(cuid())
   612→  /// Business key — unique. Maps to ERPNext item_code when synced.
   613→  itemCode        String   @unique
   614→  name            String
   615→  description     String   @default("")
   616→  /// Long-form description shown in detail modal (markdown/plain text).
   617→  longDescription String?
   618→  category        String   @default("Lainnya")
   619→  /// Price in IDR
   620→  price           Int      @default(0)
   621→  /// Old price for discount display. Null = no discount shown.
   622→  oldPrice        Int?
   623→  unit            String   @default("pcs")
   624→  stock           Int      @default(0)
   625→  emoji           String   @default("🛒")
   626→  /// Image URL (relative path like /products/tomat.png, or absolute URL)
   627→  image           String?
   628→  organic         Boolean  @default(false)
   629→  onSale          Boolean  @default(false)
   630→  /// JSON-encoded string array of tags: ["lokal","segar","organik"]
   631→  tags            String?
   632→  /// JSON-encoded string array of benefit bullet points
   633→  benefits        String?
   634→  storageTip      String?
   635→  origin          String?
   636→  published       Boolean  @default(true)
   637→  order           Int      @default(0)
   638→  /// Last ERPNext sync timestamp (null = never synced)
   639→  erpSyncedAt     DateTime?
   640→  createdAt       DateTime @default(now())
   641→  updatedAt       DateTime @updatedAt
   642→
   643→  variants        ProductVariant[]
   644→
   645→  @@index([category])
   646→  @@index([published])
   647→  @@index([order])

   679→// ============================================================
   680→// DeliverySlot — delivery time slots for scheduling.
   681→// ============================================================
   682→// Default slots generated daily: Pagi (09-12), Siang (12-15),
   683→// Sore (15-18). Admin can:
   684→//   - enable/disable scheduling (SiteSetting.deliverySchedulingEnabled)
   685→//   - block specific dates (blockedDates JSON array of ISO dates)
   686→//   - set maxOrders per slot (auto-closes when full)
   687→//   - set sameDayCutoffHour (orders before X can pick today)
   688→//
   689→// Slots are generated on-demand for the next 7 days. The
   690→// currentOrders count is maintained atomically when an order is
   691→// placed (increment) or cancelled (decrement).
   692→// ============================================================
   693→
   694→model DeliverySlot {
   695→  id            String   @id @default(cuid())
   696→  /// ISO date "YYYY-MM-DD" (the day of this slot)
   697→  date          String
   698→  /// Slot window start hour (9, 12, or 15)
   699→  startHour     Int
   700→  /// Slot window end hour (12, 15, or 18)
   701→  endHour       Int
   702→  /// Label: "Pagi" | "Siang" | "Sore"
   703→  label         String
   704→  /// Max orders for this slot. 0 = no limit.
   705→  maxOrders     Int      @default(20)
   706→  /// Current booked orders (incremented on checkout, decremented on cancel)
   707→  currentOrders Int      @default(0)
   708→  /// Admin can manually close a slot (e.g. kurir absen)
   709→  closed        Boolean  @default(false)
   710→  createdAt     DateTime @default(now())
   711→  updatedAt     DateTime @updatedAt
   712→
   713→  @@unique([date, startHour])
   714→  @@index([date])
   715→  @@index([closed])
   716→}

   719→// ============================================================
   720→// DeliveryZone — internal shipping zones with flat rates.
   721→// ============================================================
   722→// Admin defines areas they deliver to internally (e.g. BTP,
   723→// Telkomas, Makassar Pusat). Each zone has a flat shipping fee
   724→// + optional minimum order for free shipping.
   725→//
   726→// Checkout flow:
   727→//   1. Customer picks "Pengiriman Internal" → select zone
   728→//   2. System calculates shipping: zone.fee (or 0 if subtotal
   729→//      >= zone.freeShippingMin)
   730→//   3. If zone not selected → "Kirim via Gojek" (customer pays
   731→//      courier directly, system shipping = 0)
   732→// ============================================================
   733→
   734→model DeliveryZone {
   735→  id              String   @id @default(cuid())
   736→  /// Display name e.g. "BTP", "Telkomas", "Makassar Pusat"
   737→  name            String   @unique
   738→  /// Flat shipping fee in IDR
   739→  fee             Int      @default(0)
   740→  /// Minimum subtotal for free shipping. 0 = no free shipping.
   741→  freeShippingMin Int      @default(0)
   742→  /// Sort order for dropdown (lower = first)
   743→  order           Int      @default(0)
   744→  active          Boolean  @default(true)
   745→  createdAt       DateTime @default(now())
   746→  updatedAt       DateTime @updatedAt
   747→
   748→  @@index([active])
   749→}

   756→  createdAt DateTime @default(now())
