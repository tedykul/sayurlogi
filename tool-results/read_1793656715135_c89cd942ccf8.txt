    34→    34→// Session — opaque-token server sessions for admin auth.
    35→    35→// ============================================================
    36→    36→// Replaces the legacy "Bearer password" scheme. Login route
    37→    37→// mints a random 32-byte token, stores it here, and sets it as
    38→    38→// an httpOnly cookie. verifyAuth() reads the cookie → looks up
    39→    39→// the row → returns the real User (id, email, role).
    40→    40→//
    41→    41→// userId is nullable to support the legacy password-gate mode
    42→    42→// (anonymous admin session when no User row is linked). Real
    43→    43→// logins always attach a userId.
    44→    44→// ============================================================
    45→    45→
    46→    46→model Session {
    47→    47→  id        String   @id @default(cuid())
    48→    48→  token     String   @unique
    49→    49→  userId    String?
    50→    50→  user      User?    @relation(fields: [userId], references: [id], onDelete: Cascade)
    51→    51→  /// Customer (storefront) user — set when method='customer'.
    52→    52→  customerId String?
    53→    53→  customer   Customer? @relation(fields: [customerId], references: [id], onDelete: Cascade)
    54→    54→  /// Snapshot of email for display/audit when userId is null (password-gate).
    55→    55→  email     String?

    57→    57→  /// super_admin | admin | staff | viewer | password_gate | customer
    58→    58→  role      String   @default("viewer")
    59→    59→  /// "nextauth" (admin user) | "password_gate" (legacy) | "customer" (storefront)
    60→    60→  method    String   @default("nextauth")
    61→    61→  expiresAt DateTime
    62→    62→  createdAt DateTime @default(now())
    63→    63→

   487→   487→// Coupon — discount code system (customer-input codes).
   488→   488→// ============================================================
   489→   489→// Dual-layer with ERPNext Pricing Rule:
   490→   490→//   - Coupon (here): customer-input codes at checkout
   491→   491→//     (WELCOME10, ONGKIR0, HEMAT20). Validated + redeemed here.
   492→   492→//   - ERPNext Pricing Rule: automatic discounts per-item/customer
   493→   493→//     (synced into Product price). Not managed here.
   494→   494→//
   495→   495→// type: "percent" | "fixed" | "free_shipping"
   496→   496→//   percent       — value = 1..100 (percent off subtotal)
   497→   497→//   fixed         — value = IDR amount off subtotal
   498→   498→//   free_shipping — value ignored; sets shipping to 0
   499→   499→//
   500→   500→// scopeType: "all" | "category" | "product"
   501→   501→//   all       — applies to any order
   502→   502→//   category  — only when cart has items in scopeValue category
   503→   503→//   product   — only when cart has item with scopeValue itemCode
   504→   504→//
   505→   505→// Anti-abuse:
   506→   506→//   maxUsage     — total times coupon can be redeemed (0 = unlimited)
   507→   507→//   maxPerUser   — per-customer limit (identified by email or phone)
   508→   508→//   usedCount    — current redemption count (incremented atomically)
   509→   509→// ============================================================
   510→   510→
   511→   511→model Coupon {
   512→   512→  id          String   @id @default(cuid())
   513→   513→  code        String   @unique
   514→   514→  /// "percent" | "fixed" | "free_shipping"
   515→   515→  type        String
   516→   516→  /// Percent (1-100) or IDR amount. Ignored for free_shipping.
   517→   517→  value       Int      @default(0)
   518→   518→  /// Minimum subtotal (IDR) for coupon to apply. 0 = no minimum.
   519→   519→  minOrder    Int      @default(0)
   520→   520→  /// Max total redemptions. 0 = unlimited.
   521→   521→  maxUsage    Int      @default(0)
   522→   522→  /// Max redemptions per customer (by email or phone). 0 = unlimited.

   524→   524→  usedCount   Int      @default(0)
   525→   525→  startsAt    DateTime?
   526→   526→  expiresAt   DateTime?
   527→   527→  active      Boolean  @default(true)
   528→   528→  /// "all" | "category" | "product"
   529→   529→  scopeType   String   @default("all")
   530→   530→  /// Category name or itemCode (when scopeType != "all")
   531→   531→  scopeValue  String?
   532→   532→  /// Optional internal note (admin only)

   535→   535→  updatedAt   DateTime @updatedAt
   536→   536→
   537→   537→  usages      CouponUsage[]
   538→   538→
   539→   539→  @@index([active])

   680→   680→// DeliverySlot — delivery time slots for scheduling.
   681→   681→// ============================================================
   682→   682→// Default slots generated daily: Pagi (09-12), Siang (12-15),
   683→   683→// Sore (15-18). Admin can:
   684→   684→//   - enable/disable scheduling (SiteSetting.deliverySchedulingEnabled)
   685→   685→//   - block specific dates (blockedDates JSON array of ISO dates)
   686→   686→//   - set maxOrders per slot (auto-closes when full)
   687→   687→//   - set sameDayCutoffHour (orders before X can pick today)
   688→   688→//
   689→   689→// Slots are generated on-demand for the next 7 days. The
   690→   690→// currentOrders count is maintained atomically when an order is
   691→   691→// placed (increment) or cancelled (decrement).

   696→   696→  /// ISO date "YYYY-MM-DD" (the day of this slot)

   698→   698→  /// Slot window start hour (9, 12, or 15)
   699→   699→  startHour     Int
   700→   700→  /// Slot window end hour (12, 15, or 18)
   701→   701→  endHour       Int
   702→   702→  /// Label: "Pagi" | "Siang" | "Sore"
   703→   703→  label         String
   704→   704→  /// Max orders for this slot. 0 = no limit.
   705→   705→  maxOrders     Int      @default(20)
   706→   706→  /// Current booked orders (incremented on checkout, decremented on cancel)
   707→   707→  currentOrders Int      @default(0)
   708→   708→  /// Admin can manually close a slot (e.g. kurir absen)
   709→   709→  closed        Boolean  @default(false)
   710→   710→  createdAt     DateTime @default(now())
   711→   711→  updatedAt     DateTime @updatedAt
   712→   712→
   713→   713→  @@unique([date, startHour])
   714→   714→  @@index([date])
   715→   715→  @@index([closed])
